#2.anagram checking

w1 = "listen"
count1 = {}

for ch in w1:
    if ch in count1:
        count1[ch] += 1
    else:
        count1[ch] = 1

w2 = "silent"
count2 = {}

for ch in w2:
    if ch in count2:
        count2[ch] += 1
    else:
        count2[ch] = 1

if count1 == count2:
    print("anagram")
else:
    print("not anagrams")
