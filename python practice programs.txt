#2.anagram checking

w1 = "listen"
count1 = {}

for ch in w1:
    if ch in count1:
        count1[ch] += 1
    else:
        count1[ch] = 1

w2 = "silent"
count2 = {}

for ch in w2:
    if ch in count2:
        count2[ch] += 1
    else:
        count2[ch] = 1

if count1 == count2:
    print("anagram")
else:
    print("not anagrams")



#anagram using function
def check_anagram(w1, w2):
    count1 = {}
    for ch in w1:
        if ch in count1:
            count1[ch] += 1
        else:
            count1[ch] = 1

    count2 = {}
    for ch in w2:
        if ch in count2:
            count2[ch] += 1
        else:
            count2[ch] = 1

    if count1 == count2:
        print("anagram")
    else:
        print("not anagrams")

# Call the function to test it:
check_anagram("listen", "silent")



### 1. Loop Keywords (`break`, `continue`, `pass`)

```python
for n in [1, 2, 3, 4]:
    if n == 2:
        continue
    if n == 3:
        pass
    if n == 4:
        break
    print(n)

```

### 2. `return` vs `print`

```python
def add(a, b):
    return a + b

ans = add(5, 3)
print(ans)

```

### 3. `*args` and `**kwargs`

```python
def add_all(*args):
    return sum(args)

print(add_all(10, 20, 30))

def profile(**kwargs):
    print(kwargs)

profile(name="Arun", city="Salem")

```

### 4. `lambda` Function

### 5. `yield` Keyword (Generators)

```python
def get_numbers():
    yield 1
    yield 2
    yield 3

for num in get_numbers():
    print(num)

```
