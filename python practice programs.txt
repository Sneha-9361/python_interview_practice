#2.anagram checking

w1 = "listen"

for ch in w1:
    if ch in count1:
        count1[ch] += 1
    else:
        count1[ch] = 1

w2 = "silent"

    if ch in count2:
        count2[ch] += 1

    print("anagram")

    print("not anagrams")



#anagram using function
def check_anagram(w1, w2):
    count1 = {}
    for ch in w1:
        if ch in count1:
            count1[ch] += 1
        else:
            count1[ch] = 1

    count2 = {}
    for ch in w2:
        if ch in count2:
            count2[ch] += 1
        else:
            count2[ch] = 1

    if count1 == count2:
        print("anagram")
    else:
        print("not anagrams")

# Call the function to test it:
check_anagram("listen", "silent")
